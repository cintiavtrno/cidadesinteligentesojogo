if(a==='splash'){reset();render();return;}  
if(a==='resetCity'){resetCity();CITY=loadCity();reset();render();return;}

if(a==='grandFinale'){S.screen='grandFinale';render();return;}  
if(a==='charIntro'){  
const ch=el.dataset.char;  
S={screen:'charIntro',char:ch,eraIdx:0,decIdx:0,scores:{ef:5,eq:5,pr:5},choices:  
[],lastChoice:null};  
render();return;  
}
if(a==='start'){S.screen='era';render();return;}  
if(a==='choose'){  
const idx=parseInt(el.dataset.idx),ch=C().eras[S.eraIdx].decisions[S.decIdx].choices[idx];  
S.scores.ef=Math.max(1,Math.min(10,S.scores.ef+(ch.ef||0)));

S.scores.pr=Math.max(1,Math.min(10,S.scores.pr+(ch.pr||0)));  
S.lastChoice=idx;S.choices.push(idx);  
applyImpact(ch.impact);  
if((ch.pr||0)<-1||(ch.eq||0)<-1){app.classList.add('glitch-

S.screen='consequence';render();return;  
}
if(a==='nextEra'){S.eraIdx++;S.decIdx=0;S.screen='era';render();return;}  
if(a==='ending'){  
if(!CITY.playedChars.includes(S.char)){CITY.playedChars.push(S.char);saveCity(CITY);}  
S.screen='ending';render();return;  
}
});  
render();
